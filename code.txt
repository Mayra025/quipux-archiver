    from playwright.sync_api import sync_playwright

LOGIN_URL = "https://gestiondocumental.epn.edu.ec/login.php"
QUIPUX_URL = "https://gestiondocumental.epn.edu.ec/index_frames.php"

with sync_playwright() as p:

    browser = p.chromium.launch(headless=False)

    context = browser.new_context()

    page = context.new_page()

    print("Abriendo Quipux...")

    page.goto(LOGIN_URL)

    print("\n======================================")
    print("QUIPUX")
    print("======================================")
    print("Inicia sesión normalmente.")
    print("Cuando ya veas TU BANDEJA de Quipux,")
    print("regresa aquí y presiona ENTER.")
    print("======================================\n")

    input("ENTER cuando estés dentro: ")

    print("\nSesión iniciada.")
    print("Ahora vamos directamente a index_frames.php...")

    page.goto(
        QUIPUX_URL,
        wait_until="domcontentloaded"
    )

    print("\nURL actual:")
    print(page.url)

    print("\nTítulo:")
    print(page.title())

    print("\n========== TEXTO DE QUIPUX ==========\n")

    try:
        texto = page.locator("body").inner_text(
            timeout=10000
        )

        print(texto[:10000])

    except Exception as e:

        print("Error leyendo la página:")
        print(e)

    print("\n========== FRAMES ==========\n")

    for i, frame in enumerate(page.frames):

        print(f"\nFRAME {i}")
        print("URL:", frame.url)
        print("NAME:", frame.name)

        try:

            texto = frame.locator("body").inner_text(
                timeout=5000
            )

            print(texto[:3000])

        except Exception as e:

            print("No se pudo leer:", e)

    print("\n========== BÚSQUEDA AVANZADA ==========\n")

    for i, frame in enumerate(page.frames):

        try:

            cantidad = frame.get_by_text(
                "Búsqueda Avanzada",
                exact=True
            ).count()

            print(
                f"FRAME {i}: "
                f"{cantidad} coincidencia(s)"
            )

        except Exception as e:

            print(
                f"FRAME {i}: ERROR {e}"
            )

    input("\nPresiona ENTER para cerrar...")

    browser.close()

    from playwright.sync_api import sync_playwright

LOGIN_URL = "https://gestiondocumental.epn.edu.ec/login.php"
QUIPUX_URL = "https://gestiondocumental.epn.edu.ec/index_frames.php"


with sync_playwright() as p:

    browser = p.chromium.launch(headless=False)

    context = browser.new_context()

    page = context.new_page()

    print("Abriendo Quipux...")

    page.goto(LOGIN_URL)

    print("\n======================================")
    print("QUIPUX")
    print("======================================")
    print("Inicia sesión normalmente.")
    print("Cuando ya veas TU BANDEJA de Quipux,")
    print("regresa aquí y presiona ENTER.")
    print("======================================\n")

    input("ENTER cuando estés dentro: ")

    # -----------------------------------------
    # ABRIR INTERFAZ REAL
    # -----------------------------------------

    print("\nEntrando a la interfaz principal...")

    page.goto(
        QUIPUX_URL,
        wait_until="domcontentloaded"
    )

    page.wait_for_timeout(2000)

    # -----------------------------------------
    # BUSCAR FRAME DEL MENÚ
    # -----------------------------------------

    menu_frame = None

    for frame in page.frames:

        print(
            f"FRAME: {frame.name} -> {frame.url}"
        )

        if frame.name == "lieftFrame":
            menu_frame = frame

    if menu_frame is None:

        print("\n❌ No encontramos lieftFrame.")

        input("ENTER para cerrar...")
        browser.close()
        exit()

    print("\n✅ lieftFrame encontrado")

    # -----------------------------------------
    # CLIC BÚSQUEDA AVANZADA
    # -----------------------------------------

    busqueda = menu_frame.get_by_text(
        "Búsqueda Avanzada",
        exact=True
    )

    print(
        "\nCoincidencias:",
        busqueda.count()
    )

    if busqueda.count() == 0:

        print(
            "❌ No encontramos Búsqueda Avanzada."
        )

        input("ENTER para cerrar...")
        browser.close()
        exit()

    print(
        "🎯 Búsqueda Avanzada encontrada."
    )

    busqueda.click()

    # Esperar a que cambie mainFrame
    page.wait_for_timeout(2000)

    # -----------------------------------------
    # ENCONTRAR MAINFRAME ACTUAL
    # -----------------------------------------

    search_frame = None

    for frame in page.frames:

        print(
            f"\nFRAME: {frame.name}"
        )

        print(
            "URL:",
            frame.url
        )

        if (
            frame.name == "mainFrame"
            and "busquedaN" in frame.url
        ):

            search_frame = frame

    if search_frame is None:

        print(
            "\n❌ No encontramos la página de búsqueda avanzada."
        )

        input("ENTER para cerrar...")
        browser.close()
        exit()

    print("\n")
    print("======================================")
    print("🎯 BÚSQUEDA AVANZADA ENCONTRADA")
    print("======================================")

    print(
        "URL:",
        search_frame.url
    )

    # -----------------------------------------
    # INPUTS
    # -----------------------------------------

    print("\n========== INPUTS ==========\n")

    inputs = search_frame.locator("input")

    print(
        "Cantidad:",
        inputs.count()
    )

    for i in range(inputs.count()):

        elemento = inputs.nth(i)

        print(f"\nINPUT {i}")

        print(
            "type:",
            elemento.get_attribute("type")
        )

        print(
            "name:",
            elemento.get_attribute("name")
        )

        print(
            "id:",
            elemento.get_attribute("id")
        )

        print(
            "value:",
            elemento.get_attribute("value")
        )

        print(
            "placeholder:",
            elemento.get_attribute("placeholder")
        )

    # -----------------------------------------
    # SELECTS
    # -----------------------------------------

    print("\n========== SELECTS ==========\n")

    selects = search_frame.locator("select")

    print(
        "Cantidad:",
        selects.count()
    )

    for i in range(selects.count()):

        elemento = selects.nth(i)

        print(f"\nSELECT {i}")

        print(
            "name:",
            elemento.get_attribute("name")
        )

        print(
            "id:",
            elemento.get_attribute("id")
        )

        opciones = elemento.locator("option")

        for j in range(opciones.count()):

            opcion = opciones.nth(j)

            print(
                f"  [{j}] "
                f"{opcion.inner_text()!r} "
                f"value={opcion.get_attribute('value')!r}"
            )

    # -----------------------------------------
    # BOTONES
    # -----------------------------------------

    print("\n========== BOTONES ==========\n")

    botones = search_frame.locator(
        "input[type='submit'], "
        "input[type='button'], "
        "button"
    )

    print(
        "Cantidad:",
        botones.count()
    )

    for i in range(botones.count()):

        elemento = botones.nth(i)

        print(f"\nBOTÓN {i}")

        print(
            "type:",
            elemento.get_attribute("type")
        )

        print(
            "name:",
            elemento.get_attribute("name")
        )

        print(
            "id:",
            elemento.get_attribute("id")
        )

        print(
            "value:",
            elemento.get_attribute("value")
        )

    input("\nPresiona ENTER para cerrar...")

    browser.close()
